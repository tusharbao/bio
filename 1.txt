# User kadun DNA sequence input ghet aahot
# upper() sequence madhil small letters la capital letters madhe convert karto

dna = input("Enter DNA Sequence: ").upper()


# 1. LENGTH OF DNA SEQUENCE

# len() function DNA sequence madhil total characters count karto

print("\nSequence Length:", len(dna))


# 2. GC CONTENT

# DNA sequence madhun G ani C bases kiti vela aale aahet te count karto

g = dna.count("G")
c = dna.count("C")

# GC content percentage calculate karto
# G ani C chi total sankhya / Total sequence length * 100

gc = ((g + c) / len(dna)) * 100

# round(gc, 2) result decimal point nantar 2 digits paryant dakhavto

print("GC Content:", round(gc, 2), "%")


# 3. MOTIF SEARCH

# User kadun search karaycha motif input ghet aahot

motif = input("Enter Motif: ").upper()

# Motif jya positions var sapdel tya store karnyasathi empty list

positions = []

# DNA sequence madhil pratyek possible position check karto
# len(dna) - len(motif) + 1 mule last possible position suddha check hote

for i in range(len(dna) - len(motif) + 1):

    # DNA sequence madhun motif evdhach part extract karto
    # Jar extracted part motif sobat match zala tar condition true hote

    if dna[i:i + len(motif)] == motif:

        # Motif chi position list madhe add karto
        positions.append(i)

# Motif jya positions var sapadla te print karto

print("Motif Found At:", positions)


# 4. START CODON

# find() function DNA sequence madhye ATG search karto
# ATG sapadla tar tyachi position return karto
# Nahi sapadla tar -1 return karto

start = dna.find("ATG")

if start != -1:

    # Start codon sapadlyas tyachi position print karto
    print("Start Codon Found At:", start)

else:

    # Start codon sapadla nahi tar ha message print hoto
    print("Start Codon Not Found")


# 5. STOP CODONS

# DNA madhil three main stop codons chi list create keli

stop_codons = ["TAA", "TAG", "TGA"]

# List madhil pratyek stop codon check karto

for stop in stop_codons:

    # find() particular stop codon chi first position return karto

    pos = dna.find(stop)

    if pos != -1:

        # Stop codon sapadlyas tyachi position print karto
        print(stop, "Found At:", pos)

    else:

        # Stop codon sapadla nahi tar message print karto
        print(stop, "Not Found")


# 6. REVERSE COMPLEMENT

# [::-1] sequence reverse karto
# Mhanje DNA sequence cha order ulta karto

reverse = dna[::-1]

# Complement sequence store karnyasathi empty string create keli

complement = ""

# Reversed DNA sequence madhil each base check karto

for base in reverse:

    # A cha complementary base T asto
    if base == "A":
        complement += "T"

    # T cha complementary base A asto
    elif base == "T":
        complement += "A"

    # G cha complementary base C asto
    elif base == "G":
        complement += "C"

    # C cha complementary base G asto
    elif base == "C":
        complement += "G"

# Final reverse complement sequence print karto

print("Reverse Complement:", complement)
